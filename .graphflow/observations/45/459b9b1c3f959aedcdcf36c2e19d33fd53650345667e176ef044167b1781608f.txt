238:     if (std.mem.eql(u8, name, "void")) return .void_;
239:     if (std.mem.eql(u8, name, "float")) return .{ .scalar = .float };

243:     if (std.mem.eql(u8, name, "double")) return .{ .scalar = .double };

246:         .{ .prefix = "vec", .scalar = .float },

248:         .{ .prefix = "uvec", .scalar = .uint },

271: fn parseMatrixName(rest: []const u8) ?Type {
272:     if (rest.len == 0) return null;
273:     if (std.mem.indexOfScalar(u8, rest, 'x')) |split| {
274:         if (split == 0 or split + 1 >= rest.len) return null;
275:         const cols = parseSize(rest[0..split]) orelse return null;
276:         const rows = parseSize(rest[split + 1 ..]) orelse return null;
277:         if (cols < 2 or cols > 4 or rows < 2 or rows > 4) return null;
278:         return .{ .matrix = .{ .cols = cols, .rows = rows } };
279:     }
280:     const size = parseSize(rest) orelse return null;
281:     if (size < 2 or size > 4) return null;
282:     return .{ .matrix = .{ .cols = size, .rows = size } };
283: }
